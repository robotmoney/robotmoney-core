# FundState

[Git Source](https://github.com/robotmoney/robotmoney-core/blob/3dbb97a08d99eefdaa58748abc6d0f614b475645/contracts/lib/FundStorage.sol)

**Note:**

- **storage-location:** erc7201:robotmoney.storage.RobotMoneyFund

```solidity
struct FundState {
    AssetConfig[] assets;
    mapping(bytes32 codeHash => bool allowed) codeHashAllowed;
    address usdc;
    uint256 tvlCap;
    uint256 perDepositCap;
    uint256 exitFeeBps;
    address feeRecipient;
    uint256 maxSlippageBps;
    uint256 navDeviationGuardBps;
    bool depositsPaused;
    bool shutdown;
    uint64 lastStepAt;
    uint16[] targetBps;
    Plan plan;
    mapping(uint256 index => uint256) deferredTotal;
    mapping(uint256 index => mapping(address account => uint256)) deferredOf;
    bool rolesBound;
    // ── Step 3d (append-only below this line) ──
    address flowQueue;
    bool flowRoleBound;
    bool seeded;
    uint256 minSeedUsdc;
    uint16 poolShareBps;
    mapping(uint256 index => uint256) positionUnits;
    mapping(uint256 index => uint256) positionClaimTotal;
    mapping(uint256 index => mapping(address account => uint256)) positionClaimOf;
    mapping(uint256 period => uint256) inKindUsed;
    uint256 lastGoodNav;
    uint64 lastGoodNavAt;
    // ── fund-extras (append-only below this line) ──
    uint16 inKindBps;
    uint128 inKindFloorUsdc;
    uint32 inKindPeriod;
    /// @dev The cost of the step that halted the current plan (0 while it never halted, or once an extension
    ///      reopened it). A plan is over budget while its budget left is below it (review B-F5: `costSpent` never
    ///      passes `costBudget` any more, so "spent > budget" can no longer say so).
    uint256 haltCost;

    // ── step-3f (append-only strictly below `haltCost`) ─────────────────────
    uint64 depositsPausedUntil;
    uint64 lastStateStopEndedAt;
    uint64 rebalancePausedUntil;
    uint64 rebalancePauseCooldownUntil;

    uint256 minPoolDepthUsdc;
    uint16 feeMarginBps;
    uint16 delistBandBps;
    uint16 delistBudgetBps;
    uint16 maxBasketBps;

    uint256 reservedClaimUsdc;

    mapping(uint256 index => bool) inSale;
    mapping(uint256 index => uint256) saleStartValue;
    mapping(uint256 index => uint256) saleFallbackPrice;
    mapping(uint256 index => uint256) delistLossSpent;
    mapping(uint256 index => uint256) delistClaimedTokens;
    mapping(uint256 index => uint256) delistUsdcPot;
    mapping(uint256 index => uint256) delistTotalScaled;
    mapping(uint256 index => uint256) delistUsdcPerScaled;
    mapping(uint256 index => mapping(address owner => DelistClaimAccount)) delistClaims;
    mapping(uint256 index => mapping(address owner => DelistClaimAccount)) delistFeeLinked;

    mapping(uint256 index => uint256) lastGoodAssetPrice;

    mapping(uint256 index => uint256) lendingClaimEpoch;
    mapping(uint256 index => uint256) lendingUsdcPot;
    mapping(uint256 index => mapping(uint256 epoch => LendingClaimEpoch)) lendingEpochs;
    mapping(uint256 index => mapping(address owner => LendingClaimAccount)) lendingClaims;
    mapping(uint256 index => mapping(address owner => LendingClaimAccount)) lendingFeeLinked;

    address rebalancer;

    /// @dev ADR-0019: adapter runtime code hashes that are governed UUPS proxies. An adapter whose code hash is
    ///      here is vetted on every use by the code hash of the implementation it reports (`FundAdapterGuard`).
    ///      Appended (v0.9.0): the storage layout is append-only.
    mapping(bytes32 codeHash => bool isProxy) adapterProxyCodeHash;
}
```
